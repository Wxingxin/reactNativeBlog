## 7) src/features/user/userThunks.js（createAsyncThunk + condition 防重复请求）

```js
// src/features/user/userThunks.js
import { createAsyncThunk } from "@reduxjs/toolkit";
import api from "../../api";

// fetchUser(1) -> GET /users/1
export const fetchUser = createAsyncThunk(
  "user/fetch",
  async (id, thunkAPI) => {
    // 这里的 id 就是 dispatch(fetchUser(id)) 传入的参数
    const res = await api.get(`/users/${id}`);
    return res.data; // fulfilled payload
  },
  {
    // 防止重复请求：如果当前正在 loading，直接取消本次 thunk
    condition: (id, { getState }) => {
      const state = getState();
      if (state.user.status === "loading") return false;
      // 你也可以加更精细的策略：
      // if (state.user.data?.id === id && state.user.status === "succeeded") return false;
    },
  }
);
```

## 8) src/features/user/userSlice.js（pending/fulfilled/rejected + 状态机）

```js
// src/features/user/userSlice.js
import { createSlice } from "@reduxjs/toolkit";
import { fetchUser } from "./userThunks";

const initialState = {
  data: null,
  status: "idle", // idle | loading | succeeded | failed
  error: null,
};

const userSlice = createSlice({
  name: "user",
  initialState,
  reducers: {
    clearUser(state) {
      state.data = null;
      state.status = "idle";
      state.error = null;
    },
  },
  extraReducers: (builder) => {
    builder
      .addCase(fetchUser.pending, (state) => {
        state.status = "loading";
        state.error = null;
      })
      .addCase(fetchUser.fulfilled, (state, action) => {
        state.status = "succeeded";
        state.data = action.payload;
      })
      .addCase(fetchUser.rejected, (state, action) => {
        // 注意：condition 返回 false 时，不会进入 pending，也不会进入 rejected
        state.status = "failed";
        state.error = action.error?.message || "Unknown error";
      });
  },
});

export const { clearUser } = userSlice.actions;
export default userSlice.reducer;
```

## 12) src/App.jsx（React 页面：loading/成功/失败 + 防重复 + 手动触发）

```jsx
// src/App.jsx
import React, { useEffect, useState } from "react";
import { useDispatch, useSelector } from "react-redux";
import Loading from "./components/Loading";
import ErrorView from "./components/Error";
import { fetchUser } from "./features/user/userThunks";
import {
  selectUserData,
  selectUserError,
  selectUserStatus,
} from "./features/user/userSelectors";
import { clearUser } from "./features/user/userSlice";

export default function App() {
  const dispatch = useDispatch();
  const data = useSelector(selectUserData);
  const status = useSelector(selectUserStatus);
  const error = useSelector(selectUserError);

  const [userId, setUserId] = useState(1);

  // 页面首次加载/依赖变化时触发（useEffect 决定“何时发”）
  useEffect(() => {
    dispatch(fetchUser(userId));
  }, [userId, dispatch]);

  const onRetry = () => dispatch(fetchUser(userId));

  return (
    <div style={{ padding: 16, fontFamily: "system-ui, Arial" }}>
      <h1>RTK createAsyncThunk Demo</h1>

      <div style={{ display: "flex", gap: 12, alignItems: "center" }}>
        <label>
          User ID:{" "}
          <input
            type="number"
            min={1}
            max={10}
            value={userId}
            onChange={(e) => setUserId(Number(e.target.value))}
          />
        </label>

        <button onClick={() => dispatch(fetchUser(userId))}>
          Fetch (manual)
        </button>

        <button onClick={() => dispatch(clearUser())}>Clear</button>
      </div>

      <hr style={{ margin: "16px 0" }} />

      {/* 状态机渲染 */}
      {status === "loading" && <Loading />}
      {status === "failed" && <ErrorView message={error} onRetry={onRetry} />}

      {status === "succeeded" && data && (
        <div style={{ padding: 12, border: "1px solid #ddd" }}>
          <h2 style={{ marginTop: 0 }}>{data.name}</h2>
          <div>Email: {data.email}</div>
          <div>Phone: {data.phone}</div>
          <div>Website: {data.website}</div>
        </div>
      )}

      {status === "idle" && <div style={{ padding: 12 }}>No data yet.</div>}

      <hr style={{ margin: "16px 0" }} />

      <div style={{ fontSize: 12, opacity: 0.8 }}>
        Current status: <b>{status}</b>
      </div>
    </div>
  );
}
```

## 你在这个项目里已经覆盖的“必须掌握点”

* `createAsyncThunk` 自动生成 `pending/fulfilled/rejected`
* `status` 状态机设计（idle/loading/succeeded/failed）
* 页面 loading / 错误提示 / 成功渲染
* 防止重复请求：`condition`（并发控制入门级核心）
* `useEffect`（决定何时发）与 thunk（决定怎么发）的边界清晰

---

如果你要把这个例子升级成“更真实的并发安全”：**只保留最后一次请求结果（last-one-wins）**、或者 **AbortController(signal) 取消前一个请求**，我也可以在这套结构上直接改给你。
