## 6.1 上传：multipart/form-data + onUploadProgress 进度条

### 关键点你必须掌握

1. **使用 `FormData`** 承载文件与字段
2. **不要手写 `Content-Type: multipart/form-data` 的 boundary**

   * Axios/浏览器会自动加 boundary；你手写反而容易出错
3. **`onUploadProgress`** 获取上传进度（`loaded / total`）
4. 需要取消上传时可配合 `AbortController` 的 `signal`（可选）

---

### `src/api/file.js`（上传与下载 API）

```js
import { httpClient } from "./http";

/**
 * 上传头像
 * 假设接口：POST /files/avatar
 * 返回：{ url: "https://..." } 或你们的 {code,message,data}
 */
export function uploadAvatar(file, { onProgress, signal } = {}) {
  const form = new FormData();
  form.append("avatar", file); // 字段名按后端要求改
  // form.append("userId", "123"); // 需要额外字段就加

  return httpClient.post("/files/avatar", form, {
    // 不要手写 Content-Type，让浏览器自动处理 boundary
    onUploadProgress: (evt) => {
      if (!evt.total) return; // 某些环境 total 可能为空
      const percent = Math.round((evt.loaded / evt.total) * 100);
      if (onProgress) onProgress({ percent, loaded: evt.loaded, total: evt.total });
    },
    signal,
  });
}

/**
 * 下载文件（blob）
 * 假设接口：GET /files/report?id=xxx
 */
export function downloadReport(params, { signal } = {}) {
  return httpClient.get("/files/report", {
    params,
    responseType: "blob",
    signal,
    // 注意：若你的 httpClient 做了“业务解包”，这里可能需要绕开解包（后面会讲）
  });
}
```

> 注意：如果你前面 `httpClient` 的响应拦截器做了 `{code,message,data} -> data` 的“业务解包”，**下载接口通常不走 JSON**，会拿到 blob，这时需要对下载接口做“跳过解包”的策略。下面会给你两种做法（推荐做法 A）。

### `src/utils/download.js`（解析文件名 + 触发下载）

```js
// 解析 content-disposition 中的 filename / filename*
export function parseFilenameFromContentDisposition(contentDisposition) {
  if (!contentDisposition) return "";

  // 常见形式：
  // content-disposition: attachment; filename="report.pdf"
  // content-disposition: attachment; filename*=UTF-8''%E6%8A%A5%E5%91%8A.pdf
  const cd = contentDisposition;

  // 优先 filename*
  const filenameStarMatch = cd.match(/filename\*\s*=\s*UTF-8''([^;]+)/i);
  if (filenameStarMatch?.[1]) {
    try {
      return decodeURIComponent(filenameStarMatch[1].trim());
    } catch {
      return filenameStarMatch[1].trim();
    }
  }

  // fallback filename
  const filenameMatch = cd.match(/filename\s*=\s*"([^"]+)"/i) || cd.match(/filename\s*=\s*([^;]+)/i);
  if (filenameMatch?.[1]) return filenameMatch[1].trim();

  return "";
}

export function triggerBrowserDownload(blob, filename) {
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = filename || "download";
  document.body.appendChild(a);
  a.click();
  a.remove();
  URL.revokeObjectURL(url);
}
```

## 输出物 1：上传头像 + 进度条 Demo（可直接用）

### `src/pages/AvatarUploadDemo.jsx`

```jsx
import React, { useMemo, useRef, useState } from "react";
import { uploadAvatar } from "../api/file";

export default function AvatarUploadDemo() {
  const [file, setFile] = useState(null);
  const [previewUrl, setPreviewUrl] = useState("");
  const [progress, setProgress] = useState(0);
  const [status, setStatus] = useState("idle"); // idle | uploading | success | error | canceled
  const [resultUrl, setResultUrl] = useState("");

  const controllerRef = useRef(null);

  const canUpload = useMemo(() => file && status !== "uploading", [file, status]);

  function onPickFile(e) {
    const f = e.target.files?.[0];
    if (!f) return;

    setFile(f);
    setProgress(0);
    setStatus("idle");
    setResultUrl("");

    if (previewUrl) URL.revokeObjectURL(previewUrl);
    const url = URL.createObjectURL(f);
    setPreviewUrl(url);
  }

  async function onUpload() {
    if (!file) return;

    // 如有正在上传，先取消
    if (controllerRef.current) controllerRef.current.abort();
    controllerRef.current = new AbortController();

    setStatus("uploading");
    setProgress(0);

    try {
      const data = await uploadAvatar(file, {
        signal: controllerRef.current.signal,
        onProgress: ({ percent }) => setProgress(percent),
      });

      // data 是你后端返回的结果（取决于你 httpClient 的解包策略）
      // 假设 data = { url: "https://..." }
      setResultUrl(data?.url || "");
      setStatus("success");
    } catch (err) {
      const isCanceled =
        err?.name === "CanceledError" ||
        err?.name === "AbortError" ||
        String(err?.message || "").toLowerCase().includes("canceled");

      setStatus(isCanceled ? "canceled" : "error");
      console.error(err);
    }
  }

  function onCancel() {
    if (controllerRef.current) controllerRef.current.abort();
  }

  return (
    <div style={{ padding: 16, maxWidth: 520 }}>
      <h2>上传头像（multipart + 进度条）</h2>

      <input type="file" accept="image/*" onChange={onPickFile} />

      {previewUrl && (
        <div style={{ marginTop: 12 }}>
          <div style={{ marginBottom: 8 }}>预览：</div>
          <img
            src={previewUrl}
            alt="preview"
            style={{ width: 120, height: 120, borderRadius: 12, objectFit: "cover", border: "1px solid #eee" }}
          />
        </div>
      )}

      <div style={{ marginTop: 12, display: "flex", gap: 10, flexWrap: "wrap" }}>
        <button onClick={onUpload} disabled={!canUpload}>
          {status === "uploading" ? "上传中..." : "开始上传"}
        </button>
        <button onClick={onCancel} disabled={status !== "uploading"}>
          取消上传
        </button>
      </div>

      <div style={{ marginTop: 12 }}>
        <div style={{ marginBottom: 6 }}>进度：{progress}%</div>
        <div style={{ height: 10, background: "#eee", borderRadius: 999, overflow: "hidden" }}>
          <div style={{ height: "100%", width: `${progress}%`, background: "#999" }} />
        </div>
      </div>

      <div style={{ marginTop: 12 }}>
        状态：{status}
        {resultUrl ? (
          <div style={{ marginTop: 8 }}>
            上传结果 URL：<a href={resultUrl} target="_blank" rel="noreferrer">{resultUrl}</a>
          </div>
        ) : null}
      </div>
    </div>
  );
}
```

## 6.2 下载：responseType:'blob' + 解析 content-disposition + 前端触发下载

### 关键点你必须掌握

1. 请求配置 `responseType: 'blob'`
2. 从响应头中读取 `content-disposition` 获取文件名（可选）
3. `URL.createObjectURL(blob)` + `<a download>` 触发下载
4. 下载完成后 `URL.revokeObjectURL` 释放内存

---

### 下载需要注意：你现有的 `httpClient` 是否“自动解包”

如果你在响应拦截器里做了 `unwrap()`（把 `{code,message,data}` 解包），那么 blob 下载会被误处理。你有两种可靠方案：

#### 方案 A（推荐）：为下载单独建一个 raw client（不解包）

**最干净、最不容易踩坑。**

```js
// src/api/rawHttp.js
import axios from "axios";
import { getAccessToken } from "../utils/token";

const baseURL = import.meta?.env?.VITE_API_BASE_URL || "https://api.example.com";

export const rawHttp = axios.create({
  baseURL,
  timeout: 10000,
});

rawHttp.interceptors.request.use((config) => {
  const token = getAccessToken();
  if (token) {
    config.headers = config.headers || {};
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});
```

然后下载用 `rawHttp`，拿到完整响应（含 headers）：

```js
// src/api/fileDownload.js
import { rawHttp } from "./rawHttp";

export function downloadFile(url, { params, signal } = {}) {
  return rawHttp.get(url, {
    params,
    responseType: "blob",
    signal,
  }); // 返回完整 axios response
}
```

import React, { useState } from "react";
import { downloadFile } from "../api/fileDownload";
import { parseFilenameFromContentDisposition, triggerBrowserDownload } from "../utils/download";

export default function DownloadDemo() {
  const [status, setStatus] = useState("idle"); // idle | downloading | success | error

  async function onDownload() {
    setStatus("downloading");
    try {
      // 假设下载接口：GET /files/report?id=123
      const res = await downloadFile("/files/report", { params: { id: 123 } });

      const blob = res.data; // 因为 responseType: 'blob'
      const cd = res.headers?.["content-disposition"] || res.headers?.["Content-Disposition"];
      const filename = parseFilenameFromContentDisposition(cd) || "report.bin";

      triggerBrowserDownload(blob, filename);
      setStatus("success");
    } catch (err) {
      console.error(err);
      setStatus("error");
    }
  }

  return (
    <div style={{ padding: 16, maxWidth: 520 }}>
      <h2>文件下载（blob）</h2>
      <button onClick={onDownload} disabled={status === "downloading"}>
        {status === "downloading" ? "下载中..." : "下载报告"}
      </button>
      <div style={{ marginTop: 10 }}>状态：{status}</div>
    </div>
  );
}

## 大文件：分片上传（了解思路即可）

### 常见分片上传流程

1. 前端选择文件，按 `chunkSize` 切片（例如 5MB/10MB）
2. 向后端申请上传会话：得到 `uploadId`
3. 并发/串行上传每个分片：

   * 每片带上：`uploadId、chunkIndex、totalChunks、md5(可选)`
4. 上传完最后一片后调用 `complete` 合并
5. 秒传/断点续传（进阶）：

   * 计算 hash
   * 查询已上传分片列表，仅补传缺失分片

### 分片切片示意代码（核心概念）

```js
function createChunks(file, chunkSize = 5 * 1024 * 1024) {
  const chunks = [];
  let start = 0;
  let index = 0;

  while (start < file.size) {
    const end = Math.min(start + chunkSize, file.size);
    const blob = file.slice(start, end);
    chunks.push({ index, blob, start, end });
    start = end;
    index += 1;
  }

  return chunks;
}
```

### 分片上传伪代码骨架

```js
async function uploadLargeFile(file) {
  // 1) init -> uploadId
  const { uploadId } = await httpClient.post("/upload/init", {
    filename: file.name,
    size: file.size,
  });

  // 2) slice
  const chunks = createChunks(file, 10 * 1024 * 1024);

  // 3) upload each chunk（可串行；并发需要控制并发数）
  for (const c of chunks) {
    const form = new FormData();
    form.append("uploadId", uploadId);
    form.append("chunkIndex", String(c.index));
    form.append("totalChunks", String(chunks.length));
    form.append("chunk", c.blob);

    await httpClient.post("/upload/chunk", form);
  }

  // 4) complete -> server merge
  const result = await httpClient.post("/upload/complete", { uploadId });
  return result;
}
```
