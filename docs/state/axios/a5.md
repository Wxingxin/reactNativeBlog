下面给你一套**生产可用、并发安全**的 Axios Refresh Token 方案（JS/JSX），满足你列的四个核心点，并把并发控制逻辑写清楚。你按文件落地后就能直接用。

---

# 总体设计

### 关键目标

* **401 触发 refresh**
* **并发安全：同一时刻只 refresh 一次**
* **refresh 成功后，自动重放（retry）原请求**
* **refresh 失败：清理登录态 + 统一跳转登录**

### 并发控制核心思路（锁 + 队列）

* 使用一个全局变量 `refreshPromise` 作为“锁”：

  * 如果 `refreshPromise` 存在，说明正在 refresh，后续 401 不再发 refresh，而是 `await refreshPromise`
  * 如果不存在，当前 401 发起 refresh，并把 Promise 存入 `refreshPromise`
* refresh 成功后：

  * 更新 token
  * 释放锁 `refreshPromise = null`
  * 重放各自的原请求（每个请求都在 `await refreshPromise` 之后继续 retry）
* refresh 失败：

  * 清 token
  * 释放锁
  * 统一执行 `onAuthExpired()`（跳转登录）
  * 所有等待中的请求都失败返回

# 1) `src/utils/token.js`：同时管理 access_token 和 refresh_token

```js
const ACCESS_KEY = "access_token";
const REFRESH_KEY = "refresh_token";

export function getAccessToken() {
  return localStorage.getItem(ACCESS_KEY) || "";
}
export function setAccessToken(token) {
  localStorage.setItem(ACCESS_KEY, token);
}
export function clearAccessToken() {
  localStorage.removeItem(ACCESS_KEY);
}

export function getRefreshToken() {
  return localStorage.getItem(REFRESH_KEY) || "";
}
export function setRefreshToken(token) {
  localStorage.setItem(REFRESH_KEY, token);
}
export function clearRefreshToken() {
  localStorage.removeItem(REFRESH_KEY);
}

export function clearAuthTokens() {
  clearAccessToken();
  clearRefreshToken();
}
```

# 2) `src/api/auth.js`：refresh 接口（重点：用“裸 axios”避免死循环）

refresh 请求**必须**用一个不带拦截器的 axios（或同一个 axios 但明确跳过拦截器），否则 refresh 自己也 401 会触发 refresh，形成递归。

```js
import axios from "axios";
import { getRefreshToken } from "../utils/token";

const baseURL = import.meta?.env?.VITE_API_BASE_URL || "https://api.example.com";

// 裸实例：不挂你主 httpClient 的拦截器
const authClient = axios.create({
  baseURL,
  timeout: 10000,
  headers: { "Content-Type": "application/json" },
});

/**
 * 假设 refresh API:
 * POST /auth/refresh
 * body: { refreshToken }
 * 返回: { code:0, data:{ accessToken, refreshToken? }, message }
 */
export async function requestRefreshToken() {
  const refreshToken = getRefreshToken();
  if (!refreshToken) {
    const err = new Error("No refresh token");
    err.type = "auth";
    throw err;
  }

  const res = await authClient.post("/auth/refresh", { refreshToken });

  // 这里按 {code,message,data} 结构处理
  const payload = res.data;
  if (!payload || typeof payload !== "object") {
    throw new Error("Invalid refresh response");
  }
  if (payload.code !== 0) {
    const err = new Error(payload.message || "Refresh failed");
    err.type = "auth";
    err.code = payload.code;
    err.payload = payload;
    throw err;
  }

  return payload.data; // { accessToken, refreshToken? }
}
```

> 你的后端如果 refresh 返回结构不同，只改这一处即可。

# 3) `src/api/http.js`：主 httpClient + 并发安全 refresh + 重放请求

这是核心文件。

```js
import axios from "axios";
import {
  getAccessToken,
  setAccessToken,
  setRefreshToken,
  clearAuthTokens,
} from "../utils/token";
import { requestRefreshToken } from "./auth";

const baseURL = import.meta?.env?.VITE_API_BASE_URL || "https://api.example.com";

export const httpClient = axios.create({
  baseURL,
  timeout: 10000,
  headers: { "Content-Type": "application/json" },
});

// ========== 可选：统一登录失效回调 ==========
let onAuthExpired = null;
export function setOnAuthExpired(handler) {
  onAuthExpired = typeof handler === "function" ? handler : null;
}

// ========== 业务解包（可选但推荐） ==========
function unwrap(response) {
  const payload = response.data;

  // 兼容：非 {code,message,data} 直接返回
  if (!payload || typeof payload !== "object" || !("code" in payload)) {
    return payload;
  }

  const { code, message, data } = payload;
  if (code !== 0) {
    const err = new Error(message || "业务错误");
    err.type = "business";
    err.code = code;
    err.payload = payload;
    throw err;
  }

  return data;
}

// ========== 并发安全 refresh 锁 ==========
let refreshPromise = null;

// 避免无限重试：每个请求只允许 retry 一次
function markRetried(config) {
  config.__isRetryRequest = true;
  return config;
}

function isRetryRequest(config) {
  return Boolean(config?.__isRetryRequest);
}

// 判断当前请求是否是 refresh 本身（避免递归）
function isAuthRefreshRequest(config) {
  const url = config?.url || "";
  return url.includes("/auth/refresh");
}

// 执行 refresh（带锁）：同一时刻只会真正请求一次
async function getRefreshedAccessToken() {
  if (!refreshPromise) {
    refreshPromise = (async () => {
      const data = await requestRefreshToken();
      // 你的后端字段命名如果不同，这里改一下
      const newAccessToken = data.accessToken;
      const newRefreshToken = data.refreshToken;

      if (!newAccessToken) throw new Error("Refresh did not return accessToken");

      setAccessToken(newAccessToken);
      if (newRefreshToken) setRefreshToken(newRefreshToken);

      return newAccessToken;
    })().finally(() => {
      // 无论成功失败都释放锁，避免卡死
      refreshPromise = null;
    });
  }

  return refreshPromise;
}

// ========== Request Interceptor：自动携带 Token ==========
httpClient.interceptors.request.use(
  (config) => {
    const token = getAccessToken();
    if (token) {
      config.headers = config.headers || {};
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => Promise.reject(error)
);

// ========== Response Interceptor：401 refresh + 重放 ==========
httpClient.interceptors.response.use(
  (response) => unwrap(response),
  async (error) => {
    // 业务错误：直接抛出（不走 refresh）
    if (error?.type === "business") {
      return Promise.reject(error);
    }

    const status = error?.response?.status;
    const originalConfig = error?.config;

    // 非 401：原样抛出（你也可以做统一错误文案）
    if (status !== 401 || !originalConfig) {
      return Promise.reject(error);
    }

    // refresh 自己 401：直接判定登录失效，避免递归
    if (isAuthRefreshRequest(originalConfig)) {
      clearAuthTokens();
      if (onAuthExpired) onAuthExpired(error);
      return Promise.reject(error);
    }

    // 已经重试过还 401：说明 token 刷新也救不了，退出登录
    if (isRetryRequest(originalConfig)) {
      clearAuthTokens();
      if (onAuthExpired) onAuthExpired(error);
      return Promise.reject(error);
    }

    try {
      // 401 -> 触发并发安全 refresh
      const newAccessToken = await getRefreshedAccessToken();

      // 用新 token 重放原请求
      const retryConfig = markRetried({ ...originalConfig });

      retryConfig.headers = retryConfig.headers || {};
      retryConfig.headers.Authorization = `Bearer ${newAccessToken}`;

      return httpClient.request(retryConfig);
    } catch (refreshErr) {
      // refresh 失败：清理登录态 + 跳转登录 + 所有等待者失败
      clearAuthTokens();
      if (onAuthExpired) onAuthExpired(refreshErr);

      return Promise.reject(refreshErr);
    }
  }
);
```

# 并发控制逻辑说明（你可以直接写进项目文档/面试回答）

1. **401 到来时**，每个失败请求都会进入响应拦截器的 401 分支。
2. 使用 `refreshPromise` 作为“全局锁”：

   * 若 `refreshPromise` 不存在：当前请求创建 refresh Promise，并赋值给 `refreshPromise`
   * 若 `refreshPromise` 已存在：说明别的请求正在刷新，当前请求直接 `await refreshPromise`
3. refresh 成功后更新 token，并释放锁；所有等待的请求继续执行**重放原请求**。
4. refresh 失败则释放锁、清理 token、执行 `onAuthExpired()`，所有等待者都会失败返回。
5. 为避免无限循环，每个请求仅允许 retry 一次：通过 `config.__isRetryRequest` 标记。

# 常见坑与规避（这套模板已经处理）

* **refresh 请求也被拦截器拦截导致递归**：refresh 用“裸 axios”，并额外识别 `/auth/refresh`
* **多个请求同时 401 导致多次 refresh**：用 `refreshPromise` 锁
* **重试请求再次 401 无限循环**：用 `__isRetryRequest` 限制只重试一次
* **refresh 失败时未统一退出**：统一 `clearAuthTokens()` + `onAuthExpired()`
