```js
const TOKEN_KEY = "access_token";

export function getToken() {
  return localStorage.getItem(TOKEN_KEY) || "";
}

export function setToken(token) {
  localStorage.setItem(TOKEN_KEY, token);
}

export function clearToken() {
  localStorage.removeItem(TOKEN_KEY);
}
```

## 2) `src/utils/error.js`：错误文案与统一解析

这里我们把错误分为：`http / business / network / unknown`，并输出统一结构。

```js
export function parseHttpError(status) {
  if (status === 401) return "登录已过期或无权限（401）";
  if (status === 403) return "无权限访问（403）";
  if (status === 404) return "资源不存在（404）";
  if (status >= 500) return "服务器异常，请稍后重试";
  return `请求错误（${status}）`;
}

// 统一解析 axios error
export function parseAxiosError(error) {
  // 服务端返回了非 2xx
  if (error.response) {
    const status = error.response.status;
    const data = error.response.data;
    return {
      type: "http",
      message: parseHttpError(status),
      status,
      data,
    };
  }

  // 请求发出但无响应（网络、超时、CORS 等）
  if (error.request) {
    const isTimeout = String(error.message || "").toLowerCase().includes("timeout");
    return {
      type: "network",
      message: isTimeout ? "请求超时，请检查网络或稍后重试" : "网络异常，未收到服务器响应",
    };
  }

  // 其他错误（配置等）
  return { type: "unknown", message: error.message || "未知错误" };
}
```

## 3) `src/api/http.js`：axios 实例 + 拦截器（核心）

这里包含：

* `axios.create({ baseURL, timeout })`
* 请求拦截器：加 token、加公共 header
* 响应拦截器：统一解包、业务错误抛出、401 处理钩子

我默认后端返回结构是常见的：

* 成功：`{ code: 0, message: "ok", data: ... }`
* 失败：`{ code: xxx, message: "xxx", data: null }`

如果你后端不是这个结构，你把 `unwrapResponse` 改一下就行。

```js
import axios from "axios";
import { getToken, clearToken } from "../utils/token";
import { parseAxiosError } from "../utils/error";

// 你可以在 .env 里配置：VITE_API_BASE_URL
// Vite: import.meta.env.VITE_API_BASE_URL
// CRA: process.env.REACT_APP_API_BASE_URL
const baseURL = import.meta?.env?.VITE_API_BASE_URL || "https://api.example.com";

function unwrapResponse(response) {
  // 这里做“业务层解包”
  // 假设后端返回：{ code, message, data }
  const payload = response.data;

  // 兼容：有些接口可能直接返回数据数组/对象（非 {code,...}）
  if (!payload || typeof payload !== "object" || !("code" in payload)) {
    return payload;
  }

  if (payload.code !== 0) {
    const err = new Error(payload.message || "业务错误");
    err.__BUSINESS_ERROR__ = true;
    err.payload = payload;
    throw err;
  }

  return payload.data;
}

export const httpClient = axios.create({
  baseURL,
  timeout: 10000,
  headers: {
    "Content-Type": "application/json",
  },
  // validateStatus 默认即可：2xx 才进 then
});

// 请求拦截器：自动带 token
httpClient.interceptors.request.use(
  (config) => {
    const token = getToken();

    // 统一 header 注入
    if (token) {
      config.headers = config.headers || {};
      config.headers.Authorization = `Bearer ${token}`;
    }

    // 例如：统一加 trace id / client version 等
    // config.headers["X-Client"] = "web";

    return config;
  },
  (error) => Promise.reject(error)
);

// 响应拦截器：业务解包 + 错误归一
httpClient.interceptors.response.use(
  (response) => {
    // 统一返回“解包后的 data”，组件层不用再写 res.data.data
    const data = unwrapResponse(response);
    return data;
  },
  (error) => {
    // 业务错误（我们在 unwrapResponse 中手动 throw 的）
    if (error.__BUSINESS_ERROR__) {
      return Promise.reject({
        type: "business",
        message: error.message || "业务错误",
        payload: error.payload,
      });
    }

    // axios 错误统一解析
    const parsed = parseAxiosError(error);

    // 401：这里做一个“统一登出”动作（你也可以改成跳转登录）
    if (parsed.type === "http" && parsed.status === 401) {
      clearToken();
      // 这里不要直接写 window.location 也行，
      // 你可以在 UI 层监听 401 统一导航
      // window.location.href = "/login";
    }

    return Promise.reject(parsed);
  }
);
```

> 关键点：`httpClient.interceptors.response.use` 里我们把成功返回直接变成 `data`，因此业务接口会非常干净。

## 4) `src/api/user.js`：具体业务接口（只拼路径，不写拦截逻辑）

组件不写 URL，只调用这里的方法。

```js
import { httpClient } from "./http";

export const userApi = {
  // 登录：假设返回 { code:0, data:{ token } }
  login(data) {
    return httpClient.post("/auth/login", data);
  },

  // 获取个人信息：返回 profile
  getProfile() {
    return httpClient.get("/user/profile");
  },

  // 用户列表：params 示例
  list(params) {
    return httpClient.get("/users", { params });
  },

  // 创建用户：data 示例
  create(data) {
    return httpClient.post("/users", data);
  },

  // 更新用户（PUT）
  update(id, data) {
    return httpClient.put(`/users/${id}`, data);
  },

  // 局部更新（PATCH）
  patch(id, data) {
    return httpClient.patch(`/users/${id}`, data);
  },

  // 删除
  remove(id) {
    return httpClient.delete(`/users/${id}`);
  },
};
```

# 最终目标：组件里只调用 api.user.getProfile()

## 示例：`src/pages/ProfileDemo.jsx`

这里展示：如何消费统一的错误结构（http/business/network/unknown）。

```jsx
import React, { useEffect, useState } from "react";
import { api } from "../api";

export default function ProfileDemo() {
  const [profile, setProfile] = useState(null);
  const [msg, setMsg] = useState("");

  async function load() {
    setMsg("");
    try {
      const data = await api.user.getProfile();
      setProfile(data);
      setMsg("加载成功");
    } catch (err) {
      // err 已经是我们在 httpClient 里统一 Promise.reject 的结构
      if (err.type === "business") {
        setMsg(`业务错误：${err.message}`);
      } else if (err.type === "http") {
        setMsg(`HTTP错误：${err.message}`);
      } else if (err.type === "network") {
        setMsg(`网络错误：${err.message}`);
      } else {
        setMsg(`未知错误：${err?.message || "unknown"}`);
      }
    }
  }

  useEffect(() => {
    load();
  }, []);

  return (
    <div style={{ padding: 16 }}>
      <h2>Profile Demo</h2>
      <button onClick={load}>重新加载</button>
      <div style={{ marginTop: 12 }}>{msg}</div>

      <pre style={{ marginTop: 12, background: "#fafafa", padding: 12 }}>
        {profile ? JSON.stringify(profile, null, 2) : "暂无数据"}
      </pre>
    </div>
  );
}
```

# 你现在得到的关键收益

1. **组件层不写 URL、不写 token、不写 res.data 解包**
2. 所有请求统一在 `httpClient` 管理（可维护、可扩展）
3. 错误类型清晰：`business/http/network/unknown`，UI 层更好做 toast 策略
4. 后续扩展非常自然：加 refresh token、重试、日志、埋点都只改 `http.js`
