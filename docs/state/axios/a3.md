下面我给你一套在 React 项目里可直接落地的“拦截器黄金模板”，覆盖你要的 3.1 / 3.2 / 3.3，并且重点解决**拦截器叠加**这个真实工程坑。你把这些文件按路径放进去，组件层就能稳定地只写业务 API 调用。

---

# 目标设计（你最终得到什么）

1. **Request Interceptor**：自动从 localStorage 或 cookie 取 token，注入 `Authorization: Bearer xxx`
2. **Response Interceptor**：

   * 统一解包 `{ code, message, data } -> data`
   * `code != 0` 统一抛出 **业务错误**（可带 code、原始 payload）
   * HTTP `401`：统一清 token + 触发“登录失效处理”（可跳转）
   * `5xx`：统一提示“服务器异常”
3. **防叠加**：拦截器只注册一次，支持 eject，热更新不炸

## 1) `src/utils/token.js`

（沿用你已有的也可以）

```js
const TOKEN_KEY = "access_token";

export function getToken() {
  return localStorage.getItem(TOKEN_KEY) || "";
}

export function setToken(token) {
  localStorage.setItem(TOKEN_KEY, token);
}

export function clearToken() {
  localStorage.removeItem(TOKEN_KEY);
}
```

## 2) `src/utils/httpError.js`：统一错误对象与文案（推荐单独抽出）

这里我们统一输出一种错误对象结构，便于 UI 层 toast、上报。

```js
export function buildError({ type, message, status, code, payload }) {
  const err = new Error(message || "Request Error");
  err.type = type;      // "business" | "http" | "network" | "unknown"
  if (status != null) err.status = status;
  if (code != null) err.code = code;
  if (payload != null) err.payload = payload;
  return err;
}

export function httpStatusMessage(status) {
  if (status === 401) return "登录已过期或无权限（401）";
  if (status === 403) return "无权限访问（403）";
  if (status === 404) return "资源不存在（404）";
  if (status >= 500) return "服务器异常，请稍后重试";
  return `请求错误（${status}）`;
}

export function parseAxiosTransportError(error) {
  // 服务端返回非 2xx
  if (error.response) {
    const status = error.response.status;
    return buildError({
      type: "http",
      status,
      message: httpStatusMessage(status),
      payload: error.response.data,
    });
  }

  // 请求发出但无响应（网络、超时、CORS等）
  if (error.request) {
    const isTimeout = String(error.message || "").toLowerCase().includes("timeout");
    return buildError({
      type: "network",
      message: isTimeout ? "请求超时，请检查网络或稍后重试" : "网络异常，未收到服务器响应",
    });
  }

  // 其他错误
  return buildError({
    type: "unknown",
    message: error.message || "未知错误",
  });
}
```

## 3) `src/api/http.js`：核心 axios 实例 + 拦截器（含防叠加）

重点看：`attachInterceptors()` 只会注册一次，并保存 id；支持 `ejectInterceptors()`。

```js
import axios from "axios";
import { getToken, clearToken } from "../utils/token";
import { buildError, parseAxiosTransportError, httpStatusMessage } from "../utils/httpError";

// 环境变量：Vite 用 VITE_，CRA 用 REACT_APP_
// 你也可以给 baseURL 一个默认值
const baseURL = import.meta?.env?.VITE_API_BASE_URL || "https://api.example.com";

export const httpClient = axios.create({
  baseURL,
  timeout: 10000,
  headers: { "Content-Type": "application/json" },
});

// ========== 你可以对外暴露一个“登录失效回调” ==========
let onAuthExpired = null;
export function setOnAuthExpired(handler) {
  onAuthExpired = typeof handler === "function" ? handler : null;
}

// ========== 统一解包：{ code, message, data } -> data ==========
function unwrapBusinessResponse(response) {
  const payload = response.data;

  // 兼容：如果后端不是 {code,message,data}，直接返回 payload
  if (!payload || typeof payload !== "object" || !("code" in payload)) {
    return payload;
  }

  const { code, message, data } = payload;

  if (code !== 0) {
    // 业务错误：抛出统一格式
    throw buildError({
      type: "business",
      code,
      message: message || "业务错误",
      payload,
    });
  }

  return data;
}

// ========== 防叠加：保存拦截器 id ==========
const interceptorIds = {
  request: null,
  response: null,
};
let attached = false;

// 只在模块初始化调用一次（或你在入口文件手动调用一次）
export function attachInterceptors() {
  if (attached) return;
  attached = true;

  // 3.1 Request Interceptor：自动携带 Token
  interceptorIds.request = httpClient.interceptors.request.use(
    (config) => {
      const token = getToken();

      // 支持你切换 cookie token：这里仅示例 localStorage
      // 若 token 在 cookie，你就在 getToken() 内部读取即可

      if (token) {
        config.headers = config.headers || {};
        // 标准写法
        config.headers.Authorization = `Bearer ${token}`;
        // 或你公司自定义头：config.headers["X-Token"] = token;
      }

      return config;
    },
    (error) => Promise.reject(error)
  );

  // 3.2 Response Interceptor：统一解包与错误策略
  interceptorIds.response = httpClient.interceptors.response.use(
    (response) => {
      // 统一解包：让业务层拿到直接是 data
      return unwrapBusinessResponse(response);
    },
    (error) => {
      // 注意：如果 unwrapBusinessResponse 抛的是 buildError（业务错误），会直接走这里
      if (error?.type === "business") {
        return Promise.reject(error);
      }

      // 其他 axios 传输层错误统一解析
      const parsed = parseAxiosTransportError(error);

      // 401：登录失效策略
      if (parsed.type === "http" && parsed.status === 401) {
        clearToken();

        // 触发外部统一处理（例如：路由跳转、弹窗）
        if (onAuthExpired) {
          try {
            onAuthExpired(parsed);
          } catch {
            // 忽略回调内部错误，避免吞掉原始错误链路
          }
        }
      }

      // 5xx：统一提示文案（已在 httpStatusMessage 内处理）
      // parsed.message 对 5xx 会是 “服务器异常，请稍后重试”
      return Promise.reject(parsed);
    }
  );
}

// 必要时可 eject（例如在某些测试环境/微前端卸载）
export function ejectInterceptors() {
  if (!attached) return;
  attached = false;

  if (interceptorIds.request != null) {
    httpClient.interceptors.request.eject(interceptorIds.request);
    interceptorIds.request = null;
  }
  if (interceptorIds.response != null) {
    httpClient.interceptors.response.eject(interceptorIds.response);
    interceptorIds.response = null;
  }
}
```

> 关键点：**拦截器注册必须在“模块初始化处”完成，而不是写在组件里**。上面 `attachInterceptors()` 就是为了保证“只注册一次”。

# 如何在 React 项目入口“只初始化一次”（解决 3.3 叠加）

你应该在项目入口文件调用 `attachInterceptors()`，例如：

## Vite：`src/main.jsx`

```jsx
import React from "react";
import ReactDOM from "react-dom/client";
import App from "./App.jsx";
import { attachInterceptors, setOnAuthExpired } from "./api/http";

attachInterceptors();

// 统一 401 处理策略：这里给一个最常见的做法（你可以换成 react-router 的 navigate）
setOnAuthExpired(() => {
  // 例：跳登录
  window.location.href = "/login";
});

ReactDOM.createRoot(document.getElementById("root")).render(
  <React.StrictMode>
    <App />
  </React.StrictMode>
);
```

这样做的效果：

* 不会因为组件热更新、重复挂载导致拦截器叠加
* 401 统一集中处理（你也可以弹 toast、弹窗、清缓存等）

# UI 层怎么区分业务错误/网络错误/HTTP 错误（用起来像“规范”）

组件里这样用即可：

```js
try {
  const profile = await api.user.getProfile();
} catch (err) {
  if (err.type === "business") {
    // code != 0
    console.log("业务错误", err.code, err.message);
  } else if (err.type === "http") {
    console.log("HTTP错误", err.status, err.message);
  } else if (err.type === "network") {
    console.log("网络错误", err.message);
  } else {
    console.log("未知错误", err.message);
  }
}
```
