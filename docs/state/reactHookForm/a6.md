## 1) 定义 Zod Schema（单一真相：校验 + 类型）

`schema/user.schema.ts`

```ts
import { z } from "zod"

export const userSchema = z
  .object({
    username: z
      .string()
      .trim()
      .min(3, "用户名至少 3 个字符")
      .max(20, "用户名最多 20 个字符")
      .regex(/^[a-zA-Z0-9_]+$/, "只能包含字母/数字/下划线"),

    email: z
      .string()
      .trim()
      .email("邮箱格式不正确"),

    password: z
      .string()
      .min(8, "密码至少 8 位")
      .regex(/[A-Z]/, "必须包含大写字母")
      .regex(/[0-9]/, "必须包含数字"),

    confirmPassword: z.string().min(1, "请再次输入密码"),

    age: z
      .coerce // ✅ 输入框拿到的是 string，这里自动转 number
      .number()
      .int("必须是整数")
      .min(18, "年龄至少 18")
      .max(80, "年龄最多 80"),

    agree: z.boolean().refine((v) => v === true, "请勾选同意条款"),
  })
  .refine((data) => data.password === data.confirmPassword, {
    path: ["confirmPassword"], // ✅ 错误落在确认密码字段
    message: "两次输入的密码不一致",
  })

export type UserFormValues = z.infer<typeof userSchema>
```

## 2) 通用字段组件（企业项目必做：减少重复）

`components/form/FormField.tsx`

```tsx
import * as React from "react"
import type {
  FieldErrors,
  FieldValues,
  Path,
  UseFormRegister,
} from "react-hook-form"

type FormFieldProps<T extends FieldValues> = {
  label: string
  name: Path<T>
  register: UseFormRegister<T>
  errors: FieldErrors<T>
  type?: React.HTMLInputTypeAttribute
  placeholder?: string
  requiredMark?: boolean
}

export function FormField<T extends FieldValues>({
  label,
  name,
  register,
  errors,
  type = "text",
  placeholder,
  requiredMark,
}: FormFieldProps<T>) {
  const error = errors[name]?.message as string | undefined

  return (
    <div style={{ marginBottom: 14 }}>
      <label style={{ display: "block", fontWeight: 600, marginBottom: 6 }}>
        {label} {requiredMark ? <span style={{ color: "red" }}>*</span> : null}
      </label>

      <input
        {...register(name)}
        type={type}
        placeholder={placeholder}
        aria-invalid={!!error}
        style={{
          width: "100%",
          padding: "10px 12px",
          borderRadius: 8,
          border: `1px solid ${error ? "#ff4d4f" : "#d9d9d9"}`,
          outline: "none",
        }}
      />

      {error ? (
        <div style={{ color: "#ff4d4f", marginTop: 6, fontSize: 12 }}>
          {error}
        </div>
      ) : null}
    </div>
  )
}
```

> 你用 Tailwind/AntD/MUI 都行，核心是：**统一 label + input + error 展示**，以后全项目复用。

## 3) 表单页面：Zod Resolver + RHF 配置（企业级默认配置）

`pages/UserCreateForm.tsx`

```tsx
import * as React from "react"
import { useForm } from "react-hook-form"
import { zodResolver } from "@hookform/resolvers/zod"
import { userSchema, type UserFormValues } from "../schema/user.schema"
import { FormField } from "../components/form/FormField"

export default function UserCreateForm() {
  const {
    register,
    handleSubmit,
    setError,
    reset,
    watch,
    formState: { errors, isSubmitting, isDirty, isValid },
  } = useForm<UserFormValues>({
    resolver: zodResolver(userSchema),
    mode: "onBlur",         // ✅ 企业常用：失焦校验
    reValidateMode: "onChange",
    defaultValues: {
      username: "",
      email: "",
      password: "",
      confirmPassword: "",
      age: 18,
      agree: false,
    },
  })

  // （可选）实时监听：用于 UI 提示/联动
  const password = watch("password")

  const onSubmit = async (values: UserFormValues) => {
    try {
      // ✅ 模拟请求
      await fakeRequest(values)
      alert("创建成功")
      reset() // 清空
    } catch (e: any) {
      // ✅ 企业级：服务端错误映射到字段
      // 假设后端返回：{ fieldErrors: { email: "已被占用" }, message: "..." }
      const fieldErrors = e?.fieldErrors as Partial<Record<keyof UserFormValues, string>> | undefined
      if (fieldErrors) {
        Object.entries(fieldErrors).forEach(([key, msg]) => {
          setError(key as keyof UserFormValues, { type: "server", message: msg })
        })
        return
      }

      // ✅ 全局错误
      setError("root", { type: "server", message: e?.message ?? "提交失败" })
    }
  }

  return (
    <div style={{ maxWidth: 520, margin: "40px auto", padding: 20 }}>
      <h2 style={{ fontSize: 22, marginBottom: 16 }}>创建用户</h2>

      {errors.root?.message ? (
        <div style={{ background: "#fff2f0", border: "1px solid #ffccc7", padding: 12, borderRadius: 8, marginBottom: 14 }}>
          <span style={{ color: "#cf1322" }}>{errors.root.message}</span>
        </div>
      ) : null}

      <form onSubmit={handleSubmit(onSubmit)}>
        <FormField<UserFormValues>
          label="用户名"
          name="username"
          register={register}
          errors={errors}
          placeholder="例如 wjx_123"
          requiredMark
        />

        <FormField<UserFormValues>
          label="邮箱"
          name="email"
          register={register}
          errors={errors}
          placeholder="name@company.com"
          requiredMark
        />

        <FormField<UserFormValues>
          label="密码"
          name="password"
          type="password"
          register={register}
          errors={errors}
          placeholder="至少 8 位，包含大写和数字"
          requiredMark
        />

        <div style={{ marginTop: -8, marginBottom: 10, fontSize: 12, color: "#666" }}>
          当前密码长度：{password?.length ?? 0}
        </div>

        <FormField<UserFormValues>
          label="确认密码"
          name="confirmPassword"
          type="password"
          register={register}
          errors={errors}
          requiredMark
        />

        <FormField<UserFormValues>
          label="年龄"
          name="age"
          type="number"
          register={register}
          errors={errors}
          requiredMark
        />

        <div style={{ marginBottom: 14 }}>
          <label style={{ display: "flex", gap: 8, alignItems: "center" }}>
            <input type="checkbox" {...register("agree")} />
            <span>我已阅读并同意条款</span>
          </label>
          {errors.agree?.message ? (
            <div style={{ color: "#ff4d4f", marginTop: 6, fontSize: 12 }}>
              {errors.agree.message}
            </div>
          ) : null}
        </div>

        <div style={{ display: "flex", gap: 10 }}>
          <button
            type="submit"
            disabled={!isDirty || !isValid || isSubmitting}
            style={{
              padding: "10px 14px",
              borderRadius: 8,
              border: "1px solid #1677ff",
              background: (!isDirty || !isValid || isSubmitting) ? "#d6e4ff" : "#1677ff",
              color: "white",
              cursor: (!isDirty || !isValid || isSubmitting) ? "not-allowed" : "pointer",
            }}
          >
            {isSubmitting ? "提交中..." : "提交"}
          </button>

          <button
            type="button"
            onClick={() => reset()}
            disabled={isSubmitting}
            style={{
              padding: "10px 14px",
              borderRadius: 8,
              border: "1px solid #d9d9d9",
              background: "white",
              cursor: isSubmitting ? "not-allowed" : "pointer",
            }}
          >
            重置
          </button>
        </div>
      </form>
    </div>
  )
}

// ---- mock api ----
async function fakeRequest(values: UserFormValues) {
  await new Promise((r) => setTimeout(r, 600))
  if (values.email.endsWith("@test.com")) {
    const err: any = new Error("校验失败")
    err.fieldErrors = { email: "该邮箱已被占用" }
    throw err
  }
  return { ok: true }
}
```

## 4) 企业级“最佳实践清单”（你照着做就稳）

### ✅ 1) Schema 做唯一真相

* **所有字段校验写在 zod schema**
* 不要散落在 `register({ required })` 里（维护成本高）

### ✅ 2) `z.infer` 做类型自动推导

* 前端表单类型 = 后端 DTO/接口对齐的第一步

### ✅ 3) 统一 FormField / FormItem

* label、输入、错误、提示、required 星号统一
* 项目越大收益越大

### ✅ 4) 服务端错误映射（字段级）

* `setError("email", { message: "已存在" })`
* 比 toast 更友好

### ✅ 5) 交互：`mode: onBlur + reValidateMode: onChange`

* 用户体验更好（不一输入就红）
